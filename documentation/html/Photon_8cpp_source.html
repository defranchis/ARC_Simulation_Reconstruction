56Photon::DrawPhotonPath() const {
57 std::vector<std::pair<std::unique_ptr<TObject>, std::string>> PhotonLine;
58 if(!m_IsTrackDrawn) {
59 return PhotonLine;
60 }
61 const std::size_t RowToDraw = Settings::GetSizeT("EventDisplay/RowToDraw");
62 if(!m_RadiatorCell || GetRadiatorRowNumber() != RowToDraw) {
63 return PhotonLine;
64 }
65 const auto ReversePhi = m_RadiatorCell->ReversePhiRotation();
66 const auto EmissionPoint = ReversePhi(m_EmissionPoint.GlobalVector());
67 const auto Position = ReversePhi(m_Position.GlobalVector());
68 if(!m_MirrorHitPosition) {
69 TLine PhotonLine1(EmissionPoint.Z(),
70 EmissionPoint.X(),
71 Position.Z(),
72 Position.X());
73 if(m_Status == Status::MirrorMiss) {
74 PhotonLine1.SetLineColor(7);
75 } else {
76 PhotonLine1.SetLineColor(kGreen);
77 }
78 PhotonLine.push_back(std::make_pair(std::make_unique<TLine>(PhotonLine1), ""));
79 } else {
80 const auto MirrorHitPosition = ReversePhi(m_MirrorHitPosition->GlobalVector());
81 TLine PhotonLine1(EmissionPoint.Z(),
82 EmissionPoint.X(),
83 MirrorHitPosition.Z(),
84 MirrorHitPosition.X());
85 TLine PhotonLine2(MirrorHitPosition.Z(),
86 MirrorHitPosition.X(),
87 Position.Z(),
88 Position.X());
89 if(m_Status != Status::DetectorHit) {
90 PhotonLine1.SetLineColor(6);
91 PhotonLine2.SetLineColor(6);
92 } else if(m_Status == Status::DetectorMiss) {
93 PhotonLine1.SetLineColor(8);
94 PhotonLine2.SetLineColor(8);
95 } else {
96 PhotonLine1.SetLineColor(kBlue);
97 PhotonLine2.SetLineColor(kBlue);
98 }
99 PhotonLine.push_back(std::make_pair(std::make_unique<TLine>(PhotonLine1), ""));
100 PhotonLine.push_back(std::make_pair(std::make_unique<TLine>(PhotonLine2), ""));
101 }
102 return PhotonLine;
103}
