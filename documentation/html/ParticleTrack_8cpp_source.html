40bool ParticleTrack::TrackThroughTracker(const TrackingVolume &InnerTracker) {
41 if(m_Location != Location::TrackerVolume) {
42 throw std::runtime_error("Cannot track particle through inner tracker again");
43 }
44 if(Settings::GetString("General/BarrelOrEndcap") == "Barrel") {
45 const double TrackerRadius = InnerTracker.GetRadius();
46 const BarrelHelixFunctor Functor(TrackerRadius);
47 const double Tracker_s = m_Helix.SolvePathLength(Functor,
48 0.9*TrackerRadius,
49 5.0*TrackerRadius);
50 if(Tracker_s == -999.0) {
51 // Most likely the track didn't reach the barrel
52 return false;
53 }
54 const auto NewPosition = m_Helix.GetPosition(Tracker_s);
55 m_Position.SetGlobalVector(NewPosition);
56 m_PathLength = Tracker_s;
57 } else {
58 const double BarrelZ = Settings::GetDouble("ARCGeometry/BarrelZ");
59 const int Sign = m_Momentum.GlobalVector().Z() > 0 ? +1 : -1;
60 const EndCapHelixFunctor Functor(BarrelZ*Sign);
61 const double Tracker_s = m_Helix.SolvePathLength(Functor,
62 0.95*BarrelZ,
63 3.0*BarrelZ);
64 if(Tracker_s == -999.0) {
65 throw std::runtime_error("Track did not reach end cap");
66 }
67 const auto NewPosition = m_Helix.GetPosition(Tracker_s);
68 m_Position.SetGlobalVector(NewPosition);
69 m_PathLength = Tracker_s;
70 }
71 m_Location = Location::EntranceWindow;
72 return true;
73}

80void ParticleTrack::ConvertToRadiatorCoordinates() {
81 Particle::ConvertToRadiatorCoordinates();
82 // Assign local coordinates
83 const auto RadiatorPosition = m_RadiatorCell->GetRadiatorPosition();
84 const auto RadiatorRotation = m_RadiatorCell->GetRadiatorRotation();
85 m_Momentum.AssignLocalCoordinates({}, RadiatorRotation);
86 m_InitialPosition.AssignLocalCoordinates(RadiatorPosition, RadiatorRotation);
87 m_EntranceWindowPosition.AssignLocalCoordinates(RadiatorPosition,
88 RadiatorRotation);
89 // Check if particle is within acceptance
90 if(!m_RadiatorCell->IsInsideCell(m_Position) ||
91 m_Momentum.GlobalVector().Dot(m_Position.GlobalVector()) < 0.0) {
92 m_Location = Location::MissedEntranceWindow;
93 }
94 // Save the entrance window position
95 m_EntranceWindowPosition = m_Position;

106bool ParticleTrack::TrackThroughAerogel() {
107 if(!m_RadiatorCell) {
108 throw std::runtime_error("No radiator cell to track through");
109 }
110 if(m_Location != Location::EntranceWindow) {
111 throw std::runtime_error("Particle not at entrance window");
112 }
113 const bool InBarrel = Settings::GetString("General/BarrelOrEndcap") == "Barrel";
114 const ZPlaneHelixFunctor CoolingFunctor(0.0,
115 m_RadiatorCell->GetRadiatorPosition(),
116 InBarrel);
117 const double Cooling_s = m_Helix.SolvePathLength(CoolingFunctor,
118 m_PathLength,
119 1.2*m_PathLength);
120 if(Cooling_s == -999.0) {
121 return false;
122 }
123 const auto CoolingLayerPosition = m_Helix.GetPosition(Cooling_s);
124 m_Position.SetGlobalVector(CoolingLayerPosition);
125 m_PathLength = Cooling_s;
126 m_AerogelEntry_s = Cooling_s;
127 const double AerogelThickness = m_RadiatorCell->GetAerogelThickness();
128 const ZPlaneHelixFunctor AerogelFunctor(AerogelThickness,
129 m_RadiatorCell->GetRadiatorPosition(),
130 InBarrel);
131 const double Aerogel_s = m_Helix.SolvePathLength(AerogelFunctor,
132 m_PathLength,
133 1.2*m_PathLength);
134 if(Aerogel_s == -999.0) {
135 return false;
136 }
137 const auto AerogelPosition = m_Helix.GetPosition(Aerogel_s);
138 m_Position.SetGlobalVector(AerogelPosition);
139 m_PathLength = Aerogel_s;
140 m_AerogelExit_s = m_PathLength;
141 m_GasEntry_s = m_AerogelExit_s;
142 m_Location = Location::Radiator;
143 return true;
144}

146void ParticleTrack::TrackThroughGasToMirror() {
147 if(m_Location != Location::Radiator &&
148 m_Location != Location::MissedEntranceWindow) {
149 throw std::runtime_error("Cannot track through gas if not in radiator");
150 }
151 m_Position.SetGlobalVector(m_Helix.GetPosition(m_GasEntry_s));
152 m_PathLength = m_GasEntry_s;
153 const auto MirrorCentre = m_RadiatorCell->GetGlobalMirrorCentre();
154 MirrorHelixFunctor MirrorFunctor(MirrorCentre,
155 m_RadiatorCell->GetMirrorCurvature());
156 const double Mirror_s = m_Helix.SolvePathLength(MirrorFunctor,
157 m_PathLength,
158 1.5*m_PathLength);
159 if(Mirror_s == -999.0) {
160 return;
161 }
162 const auto MirrorPosition = m_Helix.GetPosition(Mirror_s);
163 m_Position.SetGlobalVector(MirrorPosition);
164 m_PathLength = Mirror_s;
165 m_GasExit_s = Mirror_s;
166 if(m_RadiatorCell->IsInsideCell(m_Position)) {
167 m_Location = Location::Mirror;
168 }

171bool ParticleTrack::TrackToNextCell(const RadiatorArray &radiatorArray) {
172 std::size_t Counter = 0;
173 while(m_Location != Location::Mirror) {
174 ConvertBackToGlobalCoordinates();
175 m_Location = Location::Radiator;
176 if(!FindRadiator(radiatorArray) || Counter > 10) {
177 return false;
178 }
179 TrackThroughGasToMirror();
180 Counter++;
181 }
182 return true;

246Photon ParticleTrack::GeneratePhoton(double Entry_s,
247 double Exit_s,
248 Photon::Radiator Radiator) const {
249 // Random emission point
250 const double RandomFraction = m_RandomEmissionPoint
251 ? Utilities::Random().Uniform(0.005, 0.995) : 0.5;
252 const double EmissionPoint_s = Entry_s + (Exit_s - Entry_s)*RandomFraction;
253 const auto EmissionPoint = m_Helix.GetPosition(EmissionPoint_s);
254 // Assumed emission point
255 const double AssumedEmissionPoint_s = 0.5*(Entry_s + Exit_s);
256 const auto AssumedEmissionPoint = m_Helix.GetPosition(AssumedEmissionPoint_s);
257 // Energy and index of refraction
258 const double Energy = Utilities::Random().Uniform(1.55, 4.31);
259 const double n_phase = Utilities::GetIndexRefraction(Radiator,
260 m_ChromaticDispersion,
261 Energy);
262 // Random azimuthal angle
263 const double phi = Utilities::Random().Uniform(0.0, 2*TMath::Pi());
264 // Cherenkov angle
265 auto GetCosTheta = [&] () {
266 double CosTheta = 1.0/(Beta()*n_phase);
267 if(CosTheta > 1.0) {
268 return 1.0;
269 } else {
270 return CosTheta;
271 }
272 };
273 // Trigonometry calculations
274 const double CosTheta = GetCosTheta();
275 const double SinTheta = TMath::Sqrt(1.0 - CosTheta*CosTheta);
276 const double CosPhi = TMath::Cos(phi);
277 const double SinPhi = (phi > TMath::Pi() ? -1.0 : +1.0)
278 *TMath::Sqrt(1.0 - CosPhi*CosPhi);
279 // Photon direction in particle rest frame
280 Vector Direction(SinTheta*CosPhi, SinTheta*SinPhi, CosTheta);
281 // Rotate to lab frame
282 const auto ParticleDirection = m_Helix.GetDirection(EmissionPoint_s);
283 const ROOT::Math::RotationY RotateY(ParticleDirection.Theta());
284 Direction = RotateY(Direction);
285 const ROOT::Math::RotationZ RotateZ(ParticleDirection.Phi());
286 Direction = RotateZ(Direction);
287 // Finally create the photon
288 return Photon(EmissionPoint,
289 AssumedEmissionPoint,
290 Direction,
291 ParticleDirection,
292 m_Helix.GetDirection(AssumedEmissionPoint_s),
293 Energy,
294 CosTheta,
295 Radiator,
296 &(*m_RadiatorCell),
297 IsTrackDrawn(),
298 1.0/m_PhotonMultiplier);
299}

301double ParticleTrack::Beta() const {
302 const double Momentum = TMath::Sqrt(m_Momentum.LocalVector().Mag2());
303 return Momentum/TMath::Sqrt(m_Mass*m_Mass + Momentum*Momentum);

306double ParticleTrack::GetPhotonYield(double x, double Beta, double n) const {
307 // TODO: Move this to separate class
308 if(Beta*n < 1.0) {
309 return 0.0;
310 } else {
311 const double Efficiency = 0.60*0.90*0.80;
312 const double DeltaE = 2.55;
313 const double SinThetaC2 = (1.0 - 1.0/TMath::Power(Beta*n, 2));
314 return x*DeltaE*37000.0*SinThetaC2*Efficiency*m_PhotonMultiplier;
315 }
316}
