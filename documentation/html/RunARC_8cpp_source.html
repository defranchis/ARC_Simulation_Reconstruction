143int main(int argc, char *argv[]) {
144 if(argc%2 != 0) {
145 std::cerr << "Usage: RunARC <run mode> <settings name> <settings filename> ...\n";
146 return 1;
147 }
148 std::cout << "Welcome to the ARC simulation and reconstruction\n";
149 for(int i = 2; i < argc; i += 2) {
150 const std::string SettingsName = argv[i];
151 const std::string SettingsFilename = argv[i + 1];
152 Settings::AddSettings(SettingsName, SettingsFilename);
153 std::cout << "Added " << SettingsName << " settings from " << SettingsFilename << "\n";
154 }
155 const std::string RunMode(argv[1]);
156 if(Settings::Exists("General/Seed")) {
157 gRandom->SetSeed(Settings::GetSizeT("General/Seed"));
158 Utilities::Random().SetSeed(Settings::GetSizeT("General/Seed"));
159 }
160 EventDisplay eventDisplay;
161 const TrackingVolume InnerTracker;
162 eventDisplay.AddObject(InnerTracker.DrawARCGeometry());
163 std::unique_ptr<RadiatorArray> radiatorArray;
164 const std::string BarrelOrEndcap = Settings::GetString("General/BarrelOrEndcap");
165 if(BarrelOrEndcap == "Barrel") {
166 radiatorArray = std::make_unique<BarrelRadiatorArray>();
167 } else if(BarrelOrEndcap == "EndCap") {
168 radiatorArray = std::make_unique<EndCapRadiatorArray>();
169 } else {
170 std::cerr << "Unknown General/BarrelOrEndcap setting \"" << BarrelOrEndcap
171 << "\", must be \"Barrel\" or \"EndCap\"\n";
172 return 1;
173 }
174 eventDisplay.AddObject(radiatorArray->DrawRadiatorArray());
175 if(RunMode == "SingleTrack") {
176 std::cout << "Run mode: Single track\n";
177 const Vector Momentum = Utilities::VectorFromSpherical(
178 Settings::GetDouble("Particle/Momentum"),
179 Settings::GetDouble("Particle/CosTheta"),
180 Settings::GetDouble("Particle/Phi"));
181 const int ParticleID = Settings::GetInt("Particle/ID");;
182 ParticleTrack particleTrack(ParticleID, Momentum,
183 0, InnerTracker.GetFieldStrength());
184 if(!particleTrack.TrackThroughTracker(InnerTracker)) {
185 std::cout << "Could not track through tracker...\n";
186 return 0;
187 }
188 if(!particleTrack.FindRadiator(*radiatorArray)) {
189 std::cout << "Track does not enter any radiator cell...\n";
190 return 0;
191 }
192 if(!particleTrack.TrackThroughAerogel()) {
193 std::cout << "Could not track through aerogel...\n";
194 return 0;
195 }
196 // Aerogel photons are generated before the track continues to the mirror
197 auto PhotonsAerogel = particleTrack.GeneratePhotonsFromAerogel();
198 particleTrack.TrackThroughGasToMirror();
199 if(particleTrack.GetParticleLocation() != ParticleTrack::Location::Mirror &&
200 !particleTrack.TrackToNextCell(*radiatorArray)) {
201 std::cout << "Track left the radiator array before reaching a mirror...\n";
202 return 0;
203 }
204 auto PhotonsGas =
205 particleTrack.GetParticleLocation() == ParticleTrack::Location::Mirror
206 ? particleTrack.GeneratePhotonsFromGas() : std::vector<Photon>();
207 std::vector<PhotonHit> photonHits;
208 // Only the photons that are detected belong in the hit map
209 for(auto &photon : PhotonsAerogel) {
210 auto photonHit = PhotonMapper::TracePhoton(photon, *radiatorArray);
211 if(photonHit && photon.GetStatus() == Photon::Status::DetectorHit) {
212 photonHits.push_back(*photonHit);
213 }
214 }
215 for(auto &photon : PhotonsGas) {
216 auto photonHit = PhotonMapper::TracePhoton(photon, *radiatorArray);
217 if(photonHit && photon.GetStatus() == Photon::Status::DetectorHit) {
218 photonHits.push_back(*photonHit);
219 }
220 }
221 std::cout << "Plotting " << photonHits.size() << " detected photon hits\n";
222 particleTrack.GetRadiatorCell()->GetDetector().PlotHits("PhotonHits.pdf", photonHits);
223 } else if(RunMode == "CherenkovAngleResolution") {
224 std::cout << "Run mode: Cherenkov angle resolution\n";
225 CherenkovFile File("CherenkovFile.root");
226 const std::size_t NumberTracks = Settings::GetSizeT("General/NumberTracks");
227 const bool Aerogel = Settings::GetString("General/GasOrAerogel") == "Aerogel";
228 const Photon::Radiator Radiator = Aerogel ?
229 Photon::Radiator::Aerogel :
230 Photon::Radiator::Gas;
231 const int Hypothesis1 = Settings::GetInt("General/MassHypothesis1");
232 const int Hypothesis2 = Settings::GetInt("General/MassHypothesis2");
233 // At least every track, so that a run with fewer than ten tracks works
234 const std::size_t ProgressStep = std::max(NumberTracks/10, std::size_t{1});
235 for(std::size_t i = 0; i < NumberTracks; i++) {
236 if(i%ProgressStep == 0) {
237 std::cout << 100*i/NumberTracks;
238 std::cout << "%\n";
239 }
240 File.NumberPhotons = 0;
241 File.NumberGoodPhotons = 0.0;
242 File.TrackNumber = i;
243 constexpr std::size_t NoCell = static_cast<std::size_t>(-1);
244 File.RadiatorRowNumber = NoCell;
245 File.RadiatorColumnNumber = NoCell;
246 // Tracks that are filled early never reach the code below, so the
247 // branches are reset to sentinels: -1 for the resolutions, the largest
248 // std::size_t for the cell indices and, since -1 m is a position a
249 // mirror hit could have, a NaN for the mirror hit
250 File.FinalRadiatorRowNumber = NoCell;
251 File.FinalRadiatorColumnNumber = NoCell;
252 const double NoMirrorHit = std::numeric_limits<double>::quiet_NaN();
253 File.MirrorHit_x = NoMirrorHit;
254 File.MirrorHit_y = NoMirrorHit;
255 File.MirrorHit_z = NoMirrorHit;
256 File.SinglePhotonResolution = -1.0;
257 File.TotalResolution = -1.0;
258 File.Significance = -1.0;
259 auto GetMomentum = [&] () {
260 if(BarrelOrEndcap == "Barrel") {
261 return Utilities::GenerateRandomBarrelTrack(File.CosTheta, File.Phi);
262 } else if(BarrelOrEndcap == "EndCap") {
263 return Utilities::GenerateRandomEndCapTrack();
264 } else {
265 return Vector(0.0, 0.0, 0.0);
266 }
267 };
268 const Vector Momentum = GetMomentum();
269 File.Momentum = TMath::Sqrt(Momentum.Mag2());
270 if(BarrelOrEndcap == "EndCap") {
271 File.CosTheta = TMath::Cos(Momentum.Theta());
272 }
273 const Vector Position(0.0, 0.0, 0.0);
274 const int ParticleID = Settings::GetInt("Particle/ID");
275 ParticleTrack particleTrack(ParticleID, Momentum,
276 i, InnerTracker.GetFieldStrength());
277 const double MomentumMag = TMath::Sqrt(Momentum.Mag2());
278 const double CherenkovAngleDifference =
279 Utilities::GetCherenkovAngleDifference(MomentumMag,
280 Hypothesis1,
281 Hypothesis2,
282 Radiator);
283 if(!particleTrack.TrackThroughTracker(InnerTracker)) {
284 continue;
285 }
286 if(!particleTrack.FindRadiator(*radiatorArray)) {
287 continue;
288 }
289 File.Phi = particleTrack.GetPosition().GlobalVector().Phi();
290 File.RadiatorRowNumber = particleTrack.GetRadiatorRowNumber();
291 File.RadiatorColumnNumber = particleTrack.GetRadiatorColumnNumber();
292 auto EntranceWindowPosition = particleTrack.GetEntranceWindowPosition();
293 File.Entrance_x = EntranceWindowPosition.X();
294 File.Entrance_y = EntranceWindowPosition.Y();
295 File.Entrance_z = EntranceWindowPosition.Z();
296 if(particleTrack.GetParticleLocation() != ParticleTrack::Location::EntranceWindow) {
297 File.ParticleLocation = particleTrack.GetParticleLocation();
298 File.Fill();
299 continue;
300 }
301 if(!particleTrack.TrackThroughAerogel()) {
302 continue;
303 }
304 std::vector<Photon> Photons;
305 if(Aerogel) {
306 Photons = particleTrack.GeneratePhotonsFromAerogel();
307 }
308 particleTrack.TrackThroughGasToMirror();
309 if(particleTrack.GetParticleLocation() != ParticleTrack::Location::Mirror) {
310 const bool HitMirror = particleTrack.TrackToNextCell(*radiatorArray);
311 if(!HitMirror) {
312 File.ParticleLocation = particleTrack.GetParticleLocation();
313 File.Fill();
314 continue;
315 }
316 }
317 if(!Aerogel) {
318 Photons = particleTrack.GeneratePhotonsFromGas();
319 }
320 File.ParticleLocation = particleTrack.GetParticleLocation();
321 File.Phi = particleTrack.GetPosition().GlobalVector().Phi();
322 auto MirrorHitPosition = particleTrack.GetPosition().GlobalVector();
323 File.MirrorHit_x = MirrorHitPosition.X();
324 File.MirrorHit_y = MirrorHitPosition.Y();
325 File.MirrorHit_z = MirrorHitPosition.Z();
326 File.FinalRadiatorRowNumber = particleTrack.GetRadiatorRowNumber();
327 File.FinalRadiatorColumnNumber = particleTrack.GetRadiatorColumnNumber();
328 eventDisplay.AddObject(particleTrack.DrawParticleTrack());
329 std::vector<double> GoodAngles;
330 if(Photons.size() > static_cast<std::size_t>(CherenkovFile::MaxPhotons)) {
331 std::cout << "Warning! Track " << i << " has " << Photons.size()
332 << " photons, only the first " << CherenkovFile::MaxPhotons
333 << " are stored\n";
334 }
335 for(auto &Photon : Photons) {
336 if(File.NumberPhotons >= CherenkovFile::MaxPhotons) {
337 break;
338 }
339 File.CherenkovAngle_True[File.NumberPhotons] =
340 TMath::ACos(Photon.GetCosCherenkovAngle());
341 auto photonHit = PhotonMapper::TracePhoton(Photon, *radiatorArray);
342 eventDisplay.AddObject(Photon.DrawPhotonPath());
343 File.HasMigrated[File.NumberPhotons] = Photon.HasPhotonMigrated() ? 1 : 0;
344 if(!Photon.GetMirrorHitPosition()) {
345 File.CherenkovAngle_Reco_TrueEmissionPoint[File.NumberPhotons] = -1.0;
346 File.CherenkovAngle_Reco[File.NumberPhotons] = -1.0;
347 } else {
348 auto reconstructedPhoton =
349 PhotonReconstructor::ReconstructPhoton(*photonHit);
350 // Failed reconstructions are stored as -1.0, like photons without a mirror hit
351 auto GetAngle = [] (double CosAngle) {
352 return PhotonReconstructor::IsReconstructed(CosAngle) ? TMath::ACos(CosAngle) : -1.0;
353 };
354 File.CherenkovAngle_Reco_TrueEmissionPoint[File.NumberPhotons] =
355 GetAngle(reconstructedPhoton.m_CosCherenkovAngle_TrueEmissionPoint);
356 File.CherenkovAngle_Reco[File.NumberPhotons] =
357 GetAngle(reconstructedPhoton.m_CosCherenkovAngle);
358 if(Photon.GetStatus() == Photon::Status::DetectorHit &&
359 File.CherenkovAngle_Reco[File.NumberPhotons] >= 0.0) {
360 File.NumberGoodPhotons += Photon.GetWeight();
361 GoodAngles.push_back(File.CherenkovAngle_Reco[File.NumberPhotons]);
362 }
363 }
364 File.PhotonEnergy[File.NumberPhotons] = Photon.GetEnergy();
365 File.PhotonStatus[File.NumberPhotons] = Photon.GetStatus();
366 const auto PhotonPosition = Photon.GetPosition().GlobalVector();
367 File.PhotonHit_x[File.NumberPhotons] = PhotonPosition.X();
368 File.PhotonHit_y[File.NumberPhotons] = PhotonPosition.Y();
369 File.PhotonHit_z[File.NumberPhotons] = PhotonPosition.Z();
370 File.NumberPhotons++;
371 }
372 const double RMS = TMath::RMS(GoodAngles.begin(), GoodAngles.end());
373 File.SinglePhotonResolution = RMS;
374 File.TotalResolution = RMS/TMath::Sqrt(File.NumberGoodPhotons);
375 File.Significance = CherenkovAngleDifference/File.TotalResolution;
376 File.Fill();
377 }
378 eventDisplay.DrawEventDisplay();
379 File.Close();
380 } else {
381 std::cerr << "Unknown run mode \"" << RunMode
382 << "\", must be \"SingleTrack\" or \"CherenkovAngleResolution\"\n";
383 return 1;
384 }
385 return 0;
386}
