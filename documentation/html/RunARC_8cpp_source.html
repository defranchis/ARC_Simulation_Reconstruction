143 if(argc%2 != 0) {
144 return 0;

146 std::cout << "Welcome to the ARC simulation and reconstruction\n";
147 for(int i = 2; i < argc; i += 2) {
148 const std::string SettingsName = argv[i];
149 const std::string SettingsFilename = argv[i + 1];
150 Settings::AddSettings(SettingsName, SettingsFilename);
151 std::cout << "Added " << SettingsName << " settings from " << SettingsFilename << "\n";
152 }
153 const std::string RunMode(argv[1]);
154 if(Settings::Exists("General/Seed")) {
155 gRandom->SetSeed(Settings::GetSizeT("General/Seed"));
156 Utilities::Random().SetSeed(Settings::GetSizeT("General/Seed"));
157 }
158 EventDisplay eventDisplay;
159 const TrackingVolume InnerTracker;
160 eventDisplay.AddObject(InnerTracker.DrawARCGeometry());
161 std::unique_ptr<RadiatorArray> radiatorArray;
162 const std::string BarrelOrEndcap = Settings::GetString("General/BarrelOrEndcap");
163 if(BarrelOrEndcap == "Barrel") {
164 radiatorArray = std::make_unique<BarrelRadiatorArray>();
165 } else if(BarrelOrEndcap == "EndCap") {
166 radiatorArray = std::make_unique<EndCapRadiatorArray>();
167 } else {
168 return 0;
169 }
170 eventDisplay.AddObject(radiatorArray->DrawRadiatorArray());
171 if(RunMode == "SingleTrack") {
172 std::cout << "Run mode: Single track\n";
173 const Vector Momentum = Utilities::VectorFromSpherical(
174 Settings::GetDouble("Particle/Momentum"),
175 Settings::GetDouble("Particle/CosTheta"),
176 Settings::GetDouble("Particle/Phi"));
177 const int ParticleID = Settings::GetInt("Particle/ID");;
178 ParticleTrack particleTrack(ParticleID, Momentum,
179 0, InnerTracker.GetFieldStrength());
180 if(!particleTrack.TrackThroughTracker(InnerTracker)) {
181 std::cout << "Could not track through tracker...\n";
182 return 0;
183 }
184 if(!particleTrack.FindRadiator(*radiatorArray)) {
185 std::cout << "Track does not enter any radiator cell...\n";
186 return 0;
187 }
188 if(!particleTrack.TrackThroughAerogel()) {
189 std::cout << "Could not track through aerogel...\n";
190 return 0;
191 }
192 // Aerogel photons are generated before the track continues to the mirror
193 auto PhotonsAerogel = particleTrack.GeneratePhotonsFromAerogel();
194 particleTrack.TrackThroughGasToMirror();
195 if(particleTrack.GetParticleLocation() != ParticleTrack::Location::Mirror &&
196 !particleTrack.TrackToNextCell(*radiatorArray)) {
197 std::cout << "Track left the radiator array before reaching a mirror...\n";
198 return 0;
199 }
200 auto PhotonsGas =
201 particleTrack.GetParticleLocation() == ParticleTrack::Location::Mirror
202 ? particleTrack.GeneratePhotonsFromGas() : std::vector<Photon>();
203 std::vector<PhotonHit> photonHits;
204 for(auto &photon : PhotonsAerogel) {
205 auto photonHit = PhotonMapper::TracePhoton(photon, *radiatorArray);
206 if(photonHit) {
207 photonHits.push_back(*photonHit);
208 }
209 }
210 for(auto &photon : PhotonsGas) {
211 auto photonHit = PhotonMapper::TracePhoton(photon, *radiatorArray);
212 if(photonHit) {
213 photonHits.push_back(*photonHit);
214 }
215 }
216 particleTrack.GetRadiatorCell()->GetDetector().PlotHits("PhotonHits.pdf", photonHits);
217 } else if(RunMode == "CherenkovAngleResolution") {
218 std::cout << "Run mode: Cherenkov angle resolution\n";
219 CherenkovFile File("CherenkovFile.root");
220 const std::size_t NumberTracks = Settings::GetSizeT("General/NumberTracks");
221 const bool Aerogel = Settings::GetString("General/GasOrAerogel") == "Aerogel";
222 const Photon::Radiator Radiator = Aerogel ?
223 Photon::Radiator::Aerogel :
224 Photon::Radiator::Gas;
225 const int Hypothesis1 = Settings::GetInt("General/MassHypothesis1");
226 const int Hypothesis2 = Settings::GetInt("General/MassHypothesis2");
227 for(std::size_t i = 0; i < NumberTracks; i++) {
228 if(i%(NumberTracks/10) == 0) {
229 std::cout << 100*i/NumberTracks;
230 std::cout << "%\n";
231 }
232 File.NumberPhotons = 0;
233 File.NumberGoodPhotons = 0.0;
234 File.TrackNumber = i;
235 File.RadiatorRowNumber = static_cast<std::size_t>(-1);
236 File.RadiatorColumnNumber = static_cast<std::size_t>(-1);
237 auto GetMomentum = [&] () {
238 if(BarrelOrEndcap == "Barrel") {
239 return Utilities::GenerateRandomBarrelTrack(File.CosTheta, File.Phi);
240 } else if(BarrelOrEndcap == "EndCap") {
241 return Utilities::GenerateRandomEndCapTrack();
242 } else {
243 return Vector(0.0, 0.0, 0.0);
244 }
245 };
246 const Vector Momentum = GetMomentum();
247 File.Momentum = TMath::Sqrt(Momentum.Mag2());
248 if(BarrelOrEndcap == "EndCap") {
249 File.CosTheta = TMath::Cos(Momentum.Theta());
250 }
251 const Vector Position(0.0, 0.0, 0.0);
252 const int ParticleID = Settings::GetInt("Particle/ID");
253 ParticleTrack particleTrack(ParticleID, Momentum,
254 i, InnerTracker.GetFieldStrength());
255 const double MomentumMag = TMath::Sqrt(Momentum.Mag2());

257 Utilities::GetCherenkovAngleDifference(MomentumMag,
258 Hypothesis1,
259 Hypothesis2,
260 Radiator);
261 if(!particleTrack.TrackThroughTracker(InnerTracker)) {
262 continue;
263 }
264 if(!particleTrack.FindRadiator(*radiatorArray)) {
265 continue;
266 }
267 File.Phi = particleTrack.GetPosition().GlobalVector().Phi();
268 File.RadiatorRowNumber = particleTrack.GetRadiatorRowNumber();
269 File.RadiatorColumnNumber = particleTrack.GetRadiatorColumnNumber();

271 File.Entrance_x = EntranceWindowPosition.X();
272 File.Entrance_y = EntranceWindowPosition.Y();
273 File.Entrance_z = EntranceWindowPosition.Z();

276 File.Fill();
277 continue;
278 }
279 if(!particleTrack.TrackThroughAerogel()) {
280 continue;
281 }
282 std::vector<Photon> Photons;
283 if(Aerogel) {
284 Photons = particleTrack.GeneratePhotonsFromAerogel();
285 }
286 particleTrack.TrackThroughGasToMirror();
287 if(particleTrack.GetParticleLocation() != ParticleTrack::Location::Mirror) {
288 const bool HitMirror = particleTrack.TrackToNextCell(*radiatorArray);
289 if(!HitMirror) {
290 File.ParticleLocation = particleTrack.GetParticleLocation();
291 File.FinalRadiatorRowNumber = static_cast<std::size_t>(-1);
292 File.FinalRadiatorColumnNumber = static_cast<std::size_t>(-1);
293 File.Fill();
294 continue;
295 }
296 }
297 if(!Aerogel) {
298 Photons = particleTrack.GeneratePhotonsFromGas();
299 }
300 File.ParticleLocation = particleTrack.GetParticleLocation();
301 File.Phi = particleTrack.GetPosition().GlobalVector().Phi();
302 auto MirrorHitPosition = particleTrack.GetPosition().GlobalVector();
303 File.MirrorHit_x = MirrorHitPosition.X();
304 File.MirrorHit_y = MirrorHitPosition.Y();
305 File.MirrorHit_z = MirrorHitPosition.Z();
306 File.FinalRadiatorRowNumber = particleTrack.GetRadiatorRowNumber();
307 File.FinalRadiatorColumnNumber = particleTrack.GetRadiatorColumnNumber();
308 eventDisplay.AddObject(particleTrack.DrawParticleTrack());
309 std::vector<double> GoodAngles;
310 if(Photons.size() > static_cast<std::size_t>(CherenkovFile::MaxPhotons)) {
311 std::cout << "Warning! Track " << i << " has " << Photons.size()
312 << " photons, only the first " << CherenkovFile::MaxPhotons
313 << " are stored\n";
314 }
315 for(auto &Photon : Photons) {
316 if(File.NumberPhotons >= CherenkovFile::MaxPhotons) {
317 break;
318 }
319 File.CherenkovAngle_True[File.NumberPhotons] =
320 TMath::ACos(Photon.GetCosCherenkovAngle());
321 auto photonHit = PhotonMapper::TracePhoton(Photon, *radiatorArray);
322 eventDisplay.AddObject(Photon.DrawPhotonPath());
323 if(!Photon.GetMirrorHitPosition()) {
324 File.CherenkovAngle_Reco_TrueEmissionPoint[File.NumberPhotons] = -1.0;
325 File.CherenkovAngle_Reco[File.NumberPhotons] = -1.0;
326 } else {
327 auto reconstructedPhoton =
328 PhotonReconstructor::ReconstructPhoton(*photonHit);
329 // Failed reconstructions are stored as -1.0, like photons without a mirror hit
330 auto GetAngle = [] (double CosAngle) {
331 return PhotonReconstructor::IsReconstructed(CosAngle) ? TMath::ACos(CosAngle) : -1.0;
332 };
333 File.CherenkovAngle_Reco_TrueEmissionPoint[File.NumberPhotons] =
334 GetAngle(reconstructedPhoton.m_CosCherenkovAngle_TrueEmissionPoint);
335 File.CherenkovAngle_Reco[File.NumberPhotons] =
336 GetAngle(reconstructedPhoton.m_CosCherenkovAngle);
337 File.HasMigrated[File.NumberPhotons] = Photon.HasPhotonMigrated() ? 1 : 0;
338 if(Photon.GetStatus() == Photon::Status::DetectorHit &&
339 File.CherenkovAngle_Reco[File.NumberPhotons] >= 0.0) {
340 File.NumberGoodPhotons += Photon.GetWeight();
341 GoodAngles.push_back(File.CherenkovAngle_Reco[File.NumberPhotons]);
342 }
343 }
344 File.PhotonEnergy[File.NumberPhotons] = Photon.GetEnergy();
345 File.PhotonStatus[File.NumberPhotons] = Photon.GetStatus();
346 const auto PhotonPosition = Photon.GetPosition().GlobalVector();
347 File.PhotonHit_x[File.NumberPhotons] = PhotonPosition.X();
348 File.PhotonHit_y[File.NumberPhotons] = PhotonPosition.Y();
349 File.PhotonHit_z[File.NumberPhotons] = PhotonPosition.Z();
350 File.NumberPhotons++;
351 }
352 const double RMS = TMath::RMS(GoodAngles.begin(), GoodAngles.end());
353 File.SinglePhotonResolution = RMS;
354 File.TotalResolution = RMS/TMath::Sqrt(File.NumberGoodPhotons);
355 File.Significance = CherenkovAngleDifference/File.TotalResolution;
356 File.Fill();
357 }
358 eventDisplay.DrawEventDisplay();
359 File.Close();
360 }
361 return 0;
362}
