17 double ReconstructCosCherenkovAngle(const PhotonHit &photonHit,
18 bool TrueEmissionPoint) {
19 const RadiatorCell *radiatorCell = photonHit.m_Photon->GetRadiatorCell();
20 auto MirrorCentre = radiatorCell->GetMirrorCentre();
21 const auto Photon = photonHit.m_Photon;
22 const Vector EmissionPoint = Photon->GetEmissionPoint(TrueEmissionPoint)
23 - MirrorCentre;
24 const Vector DetectionPoint = photonHit.m_HitPosition - MirrorCentre;
25 const double Curvature = radiatorCell->GetMirrorCurvature();
26 // Distance between emission point and mirror centre,
27 // in units of mirror curvature
28 const double EmissionMirrorDist = TMath::Sqrt(EmissionPoint.Mag2())/Curvature;
29 // Parallel component of distance between detection point and mirror centre,
30 // in units of mirror curvature
31 const double DetectionMirrorParaDist = EmissionPoint.Dot(DetectionPoint)
32 /(EmissionMirrorDist*Curvature*Curvature);
33 const double InsideSqrt = DetectionPoint.Mag2()/(Curvature*Curvature)
34 - TMath::Power(DetectionMirrorParaDist, 2);
35 if(InsideSqrt < 0.0) {
36 return FailedReconstruction;
37 }
38 const double DetectionMirrorPerpDist = TMath::Sqrt(InsideSqrt);
39 auto quarticSolution = SolveQuartic(EmissionMirrorDist,
40 DetectionMirrorParaDist,
41 DetectionMirrorPerpDist);
42 if(quarticSolution.m_DegenerateSolution) {
43 return FailedReconstruction;
44 }
45 Vector ReflectionPoint = GetReflectionPoint(EmissionPoint,
46 DetectionPoint,
47 DetectionMirrorParaDist,
48 DetectionMirrorPerpDist,
49 EmissionMirrorDist,
50 quarticSolution, 0);
51 Vector OtherReflectionPoint = GetReflectionPoint(EmissionPoint,
52 DetectionPoint,
53 DetectionMirrorParaDist,
54 DetectionMirrorPerpDist,
55 EmissionMirrorDist,
56 quarticSolution, 1);
57 if(OtherReflectionPoint.Z() > ReflectionPoint.Z()) {
58 std::swap(ReflectionPoint, OtherReflectionPoint);
59 }
60 const Vector ReflectionEmissionPoint = ReflectionPoint - EmissionPoint;
61 const Vector Direction = Photon->GetParticleDirection(TrueEmissionPoint);
62 const double CosTheta = ReflectionEmissionPoint.Unit().Dot(Direction);
63 return CosTheta;
64 }

75 QuarticSolution SolveQuartic(double EmMirrorDist,
76 double DetMirrorParaDist,
77 double DetMirrorPerpDist) {
78 // Em = Emission
79 // Det = Detection
80 const double DetMirrorDist2 = DetMirrorParaDist*DetMirrorParaDist
81 + DetMirrorPerpDist*DetMirrorPerpDist;
82 const double Denominator = 4.0*EmMirrorDist*EmMirrorDist*DetMirrorDist2;
83 const double a = -4.0*EmMirrorDist*EmMirrorDist*DetMirrorPerpDist/Denominator;
84 const double b = (DetMirrorPerpDist*DetMirrorPerpDist
85 + TMath::Power(EmMirrorDist + DetMirrorParaDist, 2))/Denominator
86 - 1.0;
87 const double c = 2.0*EmMirrorDist*DetMirrorPerpDist
88 *(EmMirrorDist - DetMirrorParaDist)/Denominator;
89 const double d = DetMirrorPerpDist*DetMirrorPerpDist
90 *(EmMirrorDist*EmMirrorDist - 1.0)/Denominator;
91 auto polySolutions = Quartic::solve_quartic(a, b, c, d);
92 QuarticSolution quarticSolution;
93 std::size_t i = 0;
94 for(const auto &polySolution : polySolutions) {
95 if(polySolution.imag() == 0.0) {
96 // Only two real solutions can be stored; any other count is degenerate
97 if(i < quarticSolution.m_SinBeta.size()) {
98 double RealPart = polySolution.real();
99 quarticSolution.m_SinBeta[i] = RealPart;
100 quarticSolution.m_CosBeta[i] = (EmMirrorDist + DetMirrorParaDist)*RealPart;
101 quarticSolution.m_CosBeta[i] += EmMirrorDist*DetMirrorPerpDist
102 *(1.0 - 2.0*RealPart*RealPart);
103 quarticSolution.m_CosBeta[i] /= DetMirrorPerpDist
104 + 2.0*EmMirrorDist*DetMirrorParaDist*RealPart;
105 }
106 i++;
107 }
108 }
109 if(i != quarticSolution.m_SinBeta.size()) {
110 quarticSolution.m_DegenerateSolution = true;
111 }
112 return quarticSolution;
113 }
