26HalfRadiatorCell::DrawRadiatorGeometry() const {
27 // First find the mirror intersections with the walls in local coordinates
28 // by solving a quadratic
29 const auto MirrorCentre = GetMirrorCentre();
30 const double b = MirrorCentre.Z();
31 const double c_left = TMath::Power(m_HexagonSize/2.0, 2)
32 + MirrorCentre.Mag2()
33 - m_MirrorCurvature*m_MirrorCurvature
34 + MirrorCentre.X()*m_HexagonSize;
35 const double c_right = MirrorCentre.Mag2()
36 - m_MirrorCurvature*m_MirrorCurvature;
37 const double s_left = TMath::Sqrt(b*b - c_left);
38 const double s_right = TMath::Sqrt(b*b - c_right);
39 const double SinLeft = s_left/m_MirrorCurvature;
40 const double SinRight = s_right/m_MirrorCurvature;
41 double LeftAngle = MirrorCentre.X() < -m_HexagonSize/2.0 ?
42 TMath::ASin(SinLeft)*180.0/TMath::Pi() :
43 180.0 - TMath::ASin(SinLeft)*180.0/TMath::Pi();
44 double RightAngle = MirrorCentre.X() > m_HexagonSize/2.0 ?
45 180.0 - TMath::ASin(SinRight)*180.0/TMath::Pi() :
46 TMath::ASin(SinRight)*180.0/TMath::Pi();
47 // Check if mirror intersects with the top wall
48 const double bb = MirrorCentre.X();
49 const double h = m_RadiatorThickness - 2.0*m_VesselThickness - m_CoolingThickness;
50 const double cc = h*h + MirrorCentre.Mag2() - 2.0*MirrorCentre.Z()*h
51 - m_MirrorCurvature*m_MirrorCurvature;
52 const double s1 = bb + TMath::Sqrt(bb*bb - cc);
53 const double s2 = bb + TMath::Sqrt(bb*bb - cc);
54 if(bb*bb > cc &&
55 (TMath::Abs(s1) < m_HexagonSize/2.0 || TMath::Abs(s2) < m_HexagonSize/2.0)) {
56 const double s = TMath::Abs(s1) < m_HexagonSize/2.0 ? s1 : s2;
57 const double MidAngle = TMath::ATan2(h - MirrorCentre.Z(),
58 s - MirrorCentre.X())*180.0/TMath::Pi();
59 if(MirrorCentre.X() < 0.0) {
60 LeftAngle = MidAngle;
61 } else {
62 RightAngle = MidAngle;
63 }
64 }
65 // Finally draw everything
66 const auto ReversePhi = ReversePhiRotation();
67 const auto MirrorCentreGlobal = ReversePhi(m_MirrorCentre.GlobalVector());
68 TArc MirrorArc(MirrorCentreGlobal.Z(),
69 MirrorCentreGlobal.X(),
70 m_MirrorCurvature,
71 RightAngle,
72 LeftAngle);
73 MirrorArc.SetLineColor(kBlack);
74 MirrorArc.SetLineWidth(2);
75 MirrorArc.SetFillColorAlpha(kWhite, 0.0);
76 std::vector<std::pair<std::unique_ptr<TObject>, std::string>> Objects;
77 Objects.push_back(std::make_pair(std::make_unique<TArc>(MirrorArc), "ONLY"));
78 // Draw the walls
79 TLine LeftLine(-m_HexagonSize/2.0 + m_Position.Z(),
80 Settings::GetDouble("ARCGeometry/Radius") - m_VesselThickness,
81 -m_HexagonSize/2.0 + m_Position.Z(),
82 Settings::GetDouble("ARCGeometry/Radius")
83 + m_RadiatorThickness - m_VesselThickness);
84 LeftLine.SetLineColor(kBlack);
85 Objects.push_back(std::make_pair(std::make_unique<TLine>(LeftLine), ""));
86 TLine RightLine(m_Position.Z(),
87 Settings::GetDouble("ARCGeometry/Radius") - m_VesselThickness,
88 m_Position.Z(),
89 Settings::GetDouble("ARCGeometry/Radius")
90 + m_RadiatorThickness - m_VesselThickness);
91 RightLine.SetLineColor(kBlack);
92 Objects.push_back(std::make_pair(std::make_unique<TLine>(RightLine), ""));
93 // Draw the detector plane
94 TLine DetectorLine(-m_HexagonSize/2.0 + m_Position.Z(),
95 Settings::GetDouble("ARCGeometry/Radius") + m_CoolingThickness,
96 m_Position.Z(),
97 Settings::GetDouble("ARCGeometry/Radius") + m_CoolingThickness);
98 DetectorLine.SetLineColor(kBlack);
99 Objects.push_back(std::make_pair(std::make_unique<TLine>(DetectorLine), ""));
100 // Draw the aerogel plane
101 TLine AerogelLine(-m_HexagonSize/2.0 + m_Position.Z(),
102 Settings::GetDouble("ARCGeometry/Radius")
103 + m_CoolingThickness + m_AerogelThickness,
104 m_Position.Z(),
105 Settings::GetDouble("ARCGeometry/Radius")
106 + m_CoolingThickness + m_AerogelThickness);
107 AerogelLine.SetLineColor(kBlack);
108 Objects.push_back(std::make_pair(std::make_unique<TLine>(AerogelLine), ""));
109 // Draw the bottom cooling plane
110 TLine CoolingLine(-m_HexagonSize/2.0 + m_Position.Z(),
111 Settings::GetDouble("ARCGeometry/Radius"),
112 m_Position.Z(),
113 Settings::GetDouble("ARCGeometry/Radius"));
114 CoolingLine.SetLineColor(kBlack);
115 Objects.push_back(std::make_pair(std::make_unique<TLine>(CoolingLine), ""));
116 // Draw SiPM
117 const auto RadiatorPosition = ReversePhi(GetRadiatorPosition());
118 Objects.push_back(std::make_pair(m_Detector.DrawSiPM(RadiatorPosition), ""));
119 return Objects;
