15BarrelRadiatorArray::BarrelRadiatorArray():
16 RadiatorArray::RadiatorArray(),
17 m_FullArray(Settings::GetBool("General/FullArray")),
18 m_BarrelRadius(Settings::GetDouble("ARCGeometry/Radius") +
19 Settings::GetDouble("RadiatorCell/CoolingThickness")),
20 m_DeltaPhi(2.0*m_yHexDist/m_BarrelRadius) {
21 if(m_FullArray) {
22 m_Cells.reserve(2*m_NumberMainRowCells);
23 for(std::size_t i = 0; i < m_NumberMainRowCells; i++) {

25 m_Cells.emplace_back(std::make_unique<BarrelRadiatorCell>(i, 1, m_xHexDist));

27 for(std::size_t i = 0; i < m_NumberMainRowCells - 1; i++) {
28 // Upper row
29 m_Cells.emplace_back(std::make_unique<BarrelRadiatorCell>(i + 1, 2, m_xHexDist));
30 }
31 m_Cells.emplace_back(std::make_unique<HalfRadiatorCell>(m_NumberMainRowCells,
32 2, m_xHexDist));
33 } else {
34 m_Cells.emplace_back(std::make_unique<BarrelRadiatorCell>(0, 0, m_xHexDist));
35 }

52const RadiatorCell* BarrelRadiatorArray::FindRadiator(Particle &particle) const {
53 if(!particle.IsAtRadiator()) {
54 throw std::runtime_error("Cannot find radiator since track is not at radiator");
55 }
56 if(m_FullArray) {
57 auto Position = particle.GetPosition().GlobalVector();
58 if(TMath::Abs(Position.Phi()) > TMath::Pi()/2.0) {
59 particle.MapPhi(TMath::Pi());
60 }
61 Position = particle.GetPosition().GlobalVector();
62 double x = Position.Z();
63 const double Radius = TMath::Sqrt(Position.X()*Position.X() +
64 Position.Y()*Position.Y());
65 // Check if particle hits outside half cell
66 if(x > 8.5*m_xHexDist) {
67 return nullptr;
68 }
69 // Account for curvature when calculating y coordinate (azimuthal)
70 double ProjectedYAngle = TMath::ASin(Position.Y()/Radius);
71 double y = m_BarrelRadius*ProjectedYAngle;
72 // First check if track is on the correct side and not hitting endcap
73 bool IsReflected = false;
74 if(x < 0.0) {
75 particle.ReflectZ();
76 IsReflected = true;
77 x = particle.GetPosition().GlobalVector().Z();
78 }
79 // Make sure track is not below the main row
80 while(IsBelowMainRow(x, y)) {
81 particle.MapPhi(+m_DeltaPhi);
82 Position = particle.GetPosition().GlobalVector();
83 ProjectedYAngle = TMath::ASin(Position.Y()/Radius);
84 y = m_BarrelRadius*ProjectedYAngle;
85 }
86 // Make sure track is not above the upper row
87 while(IsAboveUpperRow(x, y, m_BarrelRadius)) {
88 particle.MapPhi(-m_DeltaPhi);
89 Position = particle.GetPosition().GlobalVector();
90 ProjectedYAngle = TMath::ASin(Position.Y()/Radius);
91 y = m_BarrelRadius*ProjectedYAngle;
92 }
93 if(IsAboveMainRow(x, y)) {
94 // Particle hits the upper row
95 const std::size_t xIndex = static_cast<std::size_t>(x/m_xHexDist) + 1;
96 const std::size_t yIndex = 2;
97 return (*this)(xIndex, yIndex);
98 } else {
99 // Particle hits the main row
100 const std::size_t xIndex = static_cast<std::size_t>((x + m_xHexDist/2.0)/m_xHexDist);
101 const std::size_t yIndex = 1;
102 if(xIndex == 0 && yIndex == 1 && IsReflected) {
103 particle.ReflectZ();
104 }
105 return (*this)(xIndex, yIndex);
106 }
107 } else {
108 return m_Cells.begin()->get();
109 }
110}

112bool BarrelRadiatorArray::IsBelowMainRow(double x, double y) const {
113 // Count half-hexagon lengths along x
114 const std::size_t xUnits = static_cast<std::size_t>(x/(m_xHexDist/2.0));
115 // Shift x coordinate
116 const double x_shift = x - static_cast<double>(xUnits)*m_xHexDist/2.0;
117 if(xUnits%2 == 0) {
118 // If hexagon slope is upwards
119 return x_shift > TMath::Sqrt(3)*y + m_xHexDist;
120 } else {
121 // If hexagon slope is downwards
122 return x_shift < -m_xHexDist/2.0 - TMath::Sqrt(3)*y;
123 }
124}

126bool BarrelRadiatorArray::IsAboveUpperRow(double x, double y, double Radius) const {
127 // The boundary on the top is just the boundary below, but shifted
128 const double y_shift = y - m_DeltaPhi*Radius;
129 return !IsBelowMainRow(x, y_shift);

132bool BarrelRadiatorArray::IsAboveMainRow(double x, double y) const {
133 // Count half-hexagon lengths along x
134 const std::size_t xUnits = static_cast<std::size_t>(x/(m_xHexDist/2.0));
135 // Shift x coordinate
136 const double x_shift = x - static_cast<double>(xUnits)*m_xHexDist/2.0;
137 if(xUnits%2 == 0) {
138 // If hexagon slope is downwards
139 return x_shift > m_xHexDist - TMath::Sqrt(3)*y;
140 } else {
141 // If hexagon slope is downwards
142 return x_shift < TMath::Sqrt(3)*y - m_xHexDist/2.0;

146int BarrelRadiatorArray::FindRadiatorIndex(std::size_t i, std::size_t j) const {
147 if(m_FullArray) {
148 if((j != 1 && j != 2) || (i == 0 && j == 2)) {
149 throw std::invalid_argument("Radiator ("
150 + std::to_string(i)
151 + ", "
152 + std::to_string(j)
153 + ") does not exist");
154 } else {
155 if(j == 1) {
156 if(i > 8) {
157 return -1;
158 } else {
159 return static_cast<int>(i);
160 }
161 } else {

163 return -1;
164 } else {
165 return static_cast<int>(m_NumberMainRowCells + i - 1);

167 }
168 }
169 } else {

172 + std::to_string(i)
