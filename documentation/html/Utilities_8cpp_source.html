18 Vector VectorFromSpherical(double R, double CosTheta, double Phi) {
19 const double SinTheta = TMath::Sqrt(1.0 - CosTheta*CosTheta);
20 const double CosPhi = TMath::Cos(Phi);
21 const double SinPhi = TMath::Sin(Phi);
22 return Vector{R*CosPhi*SinTheta, R*SinPhi*SinTheta, R*CosTheta};
23 }

26 const double CosTheta_boundary =
27 Settings::GetDouble("ARCGeometry/CosTheta_boundary");
28 while(true) {
29 const double Radius = Settings::GetDouble("ARCGeometry/Radius");
30 const double z = gRandom->Uniform(z_min, z_max);
31 const double CosTheta = z/TMath::Sqrt(z*z + Radius*Radius);

64 Vector GenerateRandomEndCapTrack() {
65 const double CosTheta_boundary =
66 Settings::GetDouble("ARCGeometry/CosTheta_boundary");
67 const double SecTheta = 1.0/CosTheta_boundary;
68 const double TanTheta = TMath::Sqrt(SecTheta*SecTheta - 1.0);
69 const double z = Settings::GetDouble("ARCGeometry/BarrelZ");
70 const double Radius_max = TanTheta*z;
71 const double Radius_min = Settings::GetDouble("ARCGeometry/EndCapInnerRadius");
72 auto GetUniformCircle = [=] () {
73 double x = CosTheta_boundary, y = CosTheta_boundary;
74 double r2, CosTheta;
75 do {
76 x = gRandom->Uniform(-Radius_max, Radius_max);
77 y = gRandom->Uniform(-Radius_max, Radius_max);
78 r2 = x*x + y*y;
79 CosTheta = z/TMath::Sqrt(r2 + z*z);
80 } while(CosTheta < CosTheta_boundary ||
81 r2 < Radius_min*Radius_min);
82 return std::make_pair(x, y);
83 };
84 auto [x, y] = GetUniformCircle();
85 const Vector Momentum = Vector(x, y, z).Unit()*GetMomentumMag();
86 return Momentum;
87 }

103 ResolutionStruct TrackPhotons(ParticleTrack particleTrack,
104 const RadiatorCell &radiatorCell,
105 const RadiatorArray &radiatorArray) {
106 // First find the correct cell
107 if(!particleTrack.FindRadiator(radiatorArray)) {
108 return ResolutionStruct{};
109 }
110 // Track through the cell, ignoring tracks that cannot be tracked (as in apps/RunARC.cpp)
111 if(particleTrack.GetParticleLocation() != ParticleTrack::Location::EntranceWindow) {
112 return ResolutionStruct{};
113 }
114 if(!particleTrack.TrackThroughAerogel()) {
115 return ResolutionStruct{};
116 }
117 particleTrack.TrackThroughGasToMirror();
118 if(particleTrack.GetParticleLocation() != ParticleTrack::Location::Mirror) {
119 particleTrack.TrackToNextCell(radiatorArray);
120 }
121 // Check if track hit the cell we're optimising
122 if(particleTrack.GetRadiatorCell() != &radiatorCell) {
123 if(particleTrack.GetRadiatorCell()) {
124 return ResolutionStruct{};
125 } else {
126 ResolutionStruct resolutionStruct;
127 resolutionStruct.HitCorrectCell = true;
128 resolutionStruct.HitTopWall = true;
129 return resolutionStruct;
130 }
131 }
132 // If we're in the cell we're optimising, make sure we hit the mirror
133 const auto Location = particleTrack.GetParticleLocation();
134 if(Location == ParticleTrack::Location::MissedMirror) {
135 return ResolutionStruct{0.0, 0, true, true};
136 }
137 // If particle hit the mirror, generate photons from gas
138 auto Photons = Location == ParticleTrack::Location::Mirror ?
139 particleTrack.GeneratePhotonsFromGas() :
140 std::vector<Photon>();
141 // Loop over all photons
142 ResolutionStruct resolutionStruct;
143 resolutionStruct.HitCorrectCell = true;
144 std::vector<double> CherenkovAngles;
145 CherenkovAngles.reserve(Photons.size());
146 for(auto &Photon : Photons) {
147 // Trace photons through optics and obtain detector hits
148 auto photonHit = PhotonMapper::TracePhoton(Photon, radiatorArray);
149 if(Photon.GetRadiatorCell() != &radiatorCell) {
150 // If photon hits a different cell, skip
151 continue;
152 } else if(Photon.GetStatus() == Photon::Status::DetectorHit) {
153 // If photon is detected, reconstruct Cherenkov angle
154 auto reconstructedPhoton =
155 PhotonReconstructor::ReconstructPhoton(*photonHit);
156 const double CosCherenkovAngle = reconstructedPhoton.m_CosCherenkovAngle;
157 if(!PhotonReconstructor::IsReconstructed(CosCherenkovAngle)) {
158 continue;
159 }
160 const double CherenkovAngle = TMath::ACos(CosCherenkovAngle);
161 CherenkovAngles.push_back(CherenkovAngle);
162 resolutionStruct.N++;
163 resolutionStruct.CentreHitDistance += photonHit->m_CentreHitDistance;
164 } else if(Photon.GetStatus() == Photon::Status::WallMiss ||
165 Photon.GetStatus() == Photon::Status::Backwards ||
166 Photon.GetStatus() == Photon::Status::DetectorMiss) {
167 // If photon doesn't hit detector, a penalty will be added later
168 resolutionStruct.HitTopWall = true;
169 }
170 }
171 if(resolutionStruct.N > 0) {
172 resolutionStruct.x = TMath::RMS(CherenkovAngles.begin(),
173 CherenkovAngles.end());
174 resolutionStruct.x /= TMath::Sqrt(resolutionStruct.N);
175 resolutionStruct.CentreHitDistance /= resolutionStruct.N;
176 }
177 return resolutionStruct;
178 }

180 double GetIndexRefraction(Photon::Radiator Radiator,
181 bool ChromaticDispersion,
182 double Energy) {
183 switch(Radiator) {
184 case Photon::Radiator::Aerogel:
185 {
186 auto GetIndex = [] (double eph) {
187 const double InsideSqrt = 46.41/(113.8 - eph*eph) +
188 228.7/(328.5 - eph*eph);
189 return 1.0 + 0.03*2.1467*(TMath::Sqrt(1.0 + InsideSqrt) - 1.0);
190 };
191 if(ChromaticDispersion) {
192 // Equation from Roger Forty via email
193 return GetIndex(Energy);
194 } else {
195 const double eph = 1239.841987427/400.0;
196 return GetIndex(eph);
197 }
198 }
199 case Photon::Radiator::Gas:
200 {
201 auto GetIndex = [] (double L) {
202 return 1.0 + 0.25324*1e-6/((1.0/(73.7*73.7)) - (1.0/(L*L)));
203 };
204 if(ChromaticDispersion) {
205 // Pressure at 1.0 bar
206 // Sellmeier equation with coefficients from
207 // https://twiki.cern.ch/twiki/bin/view/LHCb/C4F10
208 // They are similar to A. Filippas, et al. Nucl. Instr. and Meth. B, 196 (2002),
209 // p. 340 but now quite...?
210 const double Lambda = 1239.841987427/Energy;
211 return GetIndex(Lambda);
212 } else {
213 return GetIndex(400);
214 }
215 }
216 default:
217 return 1.0;
218 }
219 }

221 double GetPredictedCherenkovAngle(double Momentum,
222 int ID,
223 Photon::Radiator Radiator) {
224 const double n = GetIndexRefraction(Radiator, false);
225 const double Mass = ParticleMass::GetMass(ID);
226 const double Energy = TMath::Sqrt(Mass*Mass + Momentum*Momentum);
227 const double v = Momentum/Energy;
228 return TMath::ACos(1.0/(v*n));
229 }

245 ULong_t TrackSeed(std::size_t Seed, std::size_t TrackNumber) {
246 // Mix the two integers (64-bit finaliser of MurmurHash3)
247 std::uint64_t x = (static_cast<std::uint64_t>(Seed) << 32) ^ TrackNumber;
248 x ^= x >> 33;
249 x *= 0xff51afd7ed558ccdULL;
250 x ^= x >> 33;
251 x *= 0xc4ceb9fe1a85ec53ULL;
252 x ^= x >> 33;
253 // TRandom3 interprets seed 0 as "seed from the clock"
254 const std::uint32_t Mixed = static_cast<std::uint32_t>(x);
255 return Mixed == 0 ? 1 : Mixed;
256 }
