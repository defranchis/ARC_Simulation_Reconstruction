35double ResolutionOptimizable::EvaluateCost(std::vector<double> x) const {

37 std::size_t j = 0;
38 for(std::size_t i = 0; i < m_Parameters.size(); i++) {
39 const auto iter = m_FixedParameters.find(i);
40 if(iter == m_FixedParameters.end()) {
41 xx[i] = x[j];
42 j++;

44 xx[i] = iter->second;

46 }

48 ResolutionUtilities::fcn(xx[0], xx[1], xx[2], xx[3], xx[4],
