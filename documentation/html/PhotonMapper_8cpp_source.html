16 double PhotonMirrorDistance(const Photon &photon) {
17 auto MirrorCentre = photon.GetRadiatorCell()->GetMirrorCentre();
18 const double R = photon.GetRadiatorCell()->GetMirrorCurvature();
19 const Vector MirrorCentreMinusPosition = MirrorCentre
20 - photon.GetPosition().LocalVector();
21 const double b = -photon.GetDirection().Dot(MirrorCentreMinusPosition);
22 const double c = (MirrorCentreMinusPosition).Mag2() - R*R;
23 if(b*b - c < 0.0) {
24 return -100.0;
25 }
26 const double Discriminant = TMath::Sqrt(b*b - c);
27 const double s1 = -b + Discriminant;
28 const double s2 = -b - Discriminant;
29 return std::max(s1, s2);
30 }

32 void TracePhotonToMirror(Photon &photon) {
33 const double s = PhotonMirrorDistance(photon);
34 photon.PropagatePhoton(s*photon.GetDirection());
35 const double MaxHeight = photon.GetRadiatorCell()->GetRadiatorThickness()
36 - 2*photon.GetRadiatorCell()->GetVesselThickness()
37 - photon.GetRadiatorCell()->GetCoolingThickness();
38 auto MirrorCentre = photon.GetRadiatorCell()->GetMirrorCentre();
39 const auto Curvature = photon.GetRadiatorCell()->GetMirrorCurvature();
40 if(photon.GetPosition().LocalVector().Z() > MaxHeight) {
41 photon.UpdatePhotonStatus(Photon::Status::WallMiss);
42 } else if(s < 0.0) {
43 photon.UpdatePhotonStatus(Photon::Status::Backwards);
44 } else if((photon.GetPosition().LocalVector() - MirrorCentre).Mag2()
45 - Curvature*Curvature > 1e-6) {
46 photon.UpdatePhotonStatus(Photon::Status::OutsideMirrorRadius);
47 } else if(photon.GetRadiatorCell()->IsInsideCell(photon)) {
48 photon.UpdatePhotonStatus(Photon::Status::MirrorHit);
49 photon.RegisterMirrorHitPosition(photon.GetPosition().GlobalVector());
50 } else {
51 photon.UpdatePhotonStatus(Photon::Status::MirrorMiss);
52 }
53 }

55 PhotonHit TracePhotonToDetector(Photon &photon) {
56 const Vector Mirror = photon.GetPosition().LocalVector();
57 auto MirrorCentre = photon.GetRadiatorCell()->GetMirrorCentre();
58 const auto Curvature = photon.GetRadiatorCell()->GetMirrorCurvature();
59 const Vector Normal = (Mirror - MirrorCentre)/Curvature;
60 photon.KickPhoton(-2*photon.GetDirection().Dot(Normal)*Normal);
61 const auto Position = photon.GetPosition().LocalVector();
62 const auto Direction = photon.GetDirection();
63 auto Detector = photon.GetRadiatorCell()->GetDetector();
64 const double Theta = Detector.GetDetectorTilt();
65 const double CosTheta = TMath::Cos(Theta);
66 const double SinTheta = (Theta < 0.0 ? -1.0 : +1.0)
67 *TMath::Sqrt(1.0 - CosTheta*CosTheta);
68 // Find intersection between straight photon trajectory and tilted detector plane
69 const double Det = SinTheta*Direction.X() - CosTheta*Direction.Z();
70 const double s = ((Position.Z() - Detector.GetDetectorZPosition())*CosTheta
71 - (Position.X() - Detector.GetDetectorXPosition())*SinTheta)/Det;
72 photon.PropagatePhoton(s*Direction);
73 // Check if hit position is inside aerogel
74 const double AerogelThickness = photon.GetRadiatorCell()->GetAerogelThickness();
75 if(photon.GetPosition().LocalVector().Z() < AerogelThickness) {
76 const double Slope = TMath::Abs(Direction.Z());
77 photon.AddAerogelTravelDistance(AerogelThickness/Slope);
78 }
79 // TODO: Account for photons generated in aerogel
80 if(IsScatteredInAerogel(photon)) {
81 photon.UpdatePhotonStatus(Photon::Status::AerogelScattered);
82 }
83 const PhotonHit photonHit = photon.GetRadiatorCell()->GetDetector().AddPhotonHit(photon);
84 return photonHit;
85 }

87 std::unique_ptr<PhotonHit> TracePhoton(Photon &photon,
88 const RadiatorArray &radiatorArray) {
89 TracePhotonToMirror(photon);
90 std::size_t Counter = 0;
91 while(photon.GetStatus() == Photon::Status::MirrorMiss && Counter < 10) {
92 photon.UpdatePhotonStatus(Photon::Status::Emitted);
93 photon.ConvertBackToGlobalCoordinates();
94 if(photon.FindRadiator(radiatorArray)) {
95 photon.PutPhotonToEmissionPoint();
96 TracePhotonToMirror(photon);
97 photon.PhotonHasMigrated();
98 } else {
99 photon.UpdatePhotonStatus(Photon::Status::MirrorMiss);
100 break;
101 }
102 Counter++;
103 }
104 if(photon.GetMirrorHitPosition()) {
105 PhotonHit photonHit = TracePhotonToDetector(photon);
106 return std::make_unique<PhotonHit>(photonHit);
107 } else {
108 return nullptr;

112 bool IsScatteredInAerogel(const Photon &photon) {
113 constexpr double T0 = 0.9679;
114 constexpr double Clarity = 5.087e10;
115 const double Lambda = 1239.8/photon.GetEnergy();
116 const double Lambda2 = Lambda*Lambda;
117 const double Lambda4 = Lambda2*Lambda2;
118 const double Exponent = -Clarity*(photon.GetAerogelTravelDistance())/Lambda4;
119 const double Transmission = T0*TMath::Exp(Exponent);
120 return Utilities::Random().Uniform(0.0, 1.0) > Transmission;
121 }
