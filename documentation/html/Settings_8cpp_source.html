74 std::vector<std::string> SplitList(const std::string &Setting,
75 const std::string &Line) {
76 std::vector<std::string> Entries;
77 std::size_t Start = 0;
78 while(true) {
79 const std::size_t Comma = Line.find(',', Start);
80 const std::size_t Length =
81 Comma == std::string::npos ? std::string::npos : Comma - Start;
82 const std::string Entry = Trim(Line.substr(Start, Length));
83 if(Entry.empty()) {
84 throw std::runtime_error("Empty entry in the list "
85 + SettingName(Setting));
86 }
87 if(Entry.find_first_of(" \t") != std::string::npos) {
88 throw std::runtime_error("Entry '" + Entry + "' of the list "
89 + SettingName(Setting)
90 + " contains whitespace, separate the entries"
91 + " with commas");
92 }
93 Entries.push_back(Entry);
94 if(Comma == std::string::npos) {
95 return Entries;
96 }
97 Start = Comma + 1;
98 }
99 }

102void Settings::AddSettings(const std::string &Name, const std::string &Filename) {
103 if(m_Settings.find(Name) != m_Settings.end()) {
104 throw std::runtime_error("Settings " + Name + " already exists");
105 }
106 ssMap NewSettings;
107 std::ifstream File(Filename);
108 if(!File.is_open()) {
109 throw std::runtime_error("Cannot open settings file " + Filename
110 + " for settings " + Name);
111 }
112 std::string Line;
113 std::size_t LineNumber = 0;
114 while(std::getline(File, Line)) {
115 LineNumber++;
116 Line = Line.substr(0, Line.find('#'));
117 std::stringstream ss(Line);
118 std::string Key;
119 if(!(ss >> Key)) {
120 // Blank line, or a line that is only a comment
121 continue;
122 }
123 // Store the rest of the line, since a value may be a list with whitespace
124 std::string Value;
125 std::getline(ss >> std::ws, Value);
126 if(Value.empty()) {
127 throw std::runtime_error("Key " + Key + " on line "
128 + std::to_string(LineNumber) + " of settings file "
129 + Filename + " has no value");
130 }
131 if(NewSettings.find(Key) != NewSettings.end()) {
132 throw std::runtime_error("Key " + Key + " in settings " + Name + " already exists");
133 }
134 NewSettings.insert({Key, Value});
135 }
136 File.close();
137 m_Settings.insert({Name, std::move(NewSettings)});
138}

146std::string Settings::GetRawString(const std::string &Setting) {
147 const std::size_t SlashPos = Setting.find('/');
148 std::string Name = Setting.substr(0, SlashPos);
149 auto iter1 = m_Settings.find(Name);
150 if(iter1 == m_Settings.end()) {
151 std::string Known;
152 for(const auto &[KnownName, KnownSettings] : m_Settings) {
153 Known += " " + KnownName;
154 }
155 throw std::runtime_error("Cannot find settings name " + Name
156 + ", the settings added are:" + Known);
157 }
158 std::string Key = Setting.substr(SlashPos + 1);
159 auto iter2 = iter1->second.find(Key);
160 if(iter2 == iter1->second.end()) {
161 throw std::runtime_error("Cannot find key " + Key + " in settings " + Name);
162 }
163 return iter2->second;
164}
