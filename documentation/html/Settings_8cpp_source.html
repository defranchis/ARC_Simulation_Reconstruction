16void Settings::AddSettings(const std::string &Name, const std::string &Filename) {
17 if(m_Settings.find(Name) != m_Settings.end()) {
18 throw std::runtime_error("Settings " + Name + " already exists");
19 }
20 ssMap NewSettings;
21 std::ifstream File(Filename);
22 std::string Line;
23 while(std::getline(File, Line)) {
24 Line = Line.substr(0, Line.find('#'));
25 if(Line.empty()) {
26 continue;
27 }
28 std::stringstream ss(Line);
29 std::string Key, Value;
30 ss >> Key >> Value;
31 if(NewSettings.find(Key) != NewSettings.end()) {
32 throw std::runtime_error("Key " + Key + " in settings " + Name + " already exists");
33 }
34 NewSettings.insert({Key, Value});
35 }
36 File.close();
37 m_Settings.insert({Name, std::move(NewSettings)});
38}

40std::string Settings::GetString(const std::string &Setting) {
41 const std::size_t SlashPos = Setting.find('/');
42 std::string Name = Setting.substr(0, SlashPos);
43 auto iter1 = m_Settings.find(Name);
44 if(iter1 == m_Settings.end()) {
45 throw std::runtime_error("Cannot find settings name " + Name);

47 std::string Key = Setting.substr(SlashPos + 1);
48 auto iter2 = iter1->second.find(Key);
