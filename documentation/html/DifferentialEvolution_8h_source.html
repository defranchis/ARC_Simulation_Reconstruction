118 void InitPopulation()
119 {
120 // Init population based on random sampling of the cost function
121 std::shared_ptr<std::uniform_real_distribution<double>> distribution;
122
123 for (auto& agent : m_population)
124 {
125 for (std::size_t i = 0; i < m_numberOfParameters; i++)
126 {
127 if (m_constraints[i].isConstrained)
128 {
129 distribution = std::make_shared<std::uniform_real_distribution<double>>(std::uniform_real_distribution<double>(m_constraints[i].lower, m_constraints[i].upper));
130 }
131 else
132 {
133 distribution = std::make_shared<std::uniform_real_distribution<double>>(std::uniform_real_distribution<double>(g_defaultLowerConstraint, g_defaultUpperConstarint));
134 }
135
136 agent[i] = (*distribution)(m_generator);
137 }
138 }
139
140 // Initialize minimum cost, best agent and best agent index
141 for (std::size_t i = 0; i < m_populationSize; i++)
142 {
143 m_minCostPerAgent[i] = m_cost.EvaluateCost(m_population[i]);
144
145 if (m_minCostPerAgent[i] < m_minCost)
146 {
147 m_minCost = m_minCostPerAgent[i];
148 m_bestAgentIndex = i;
149 }

153 void SelectionAndCorssing()
154 {
155 std::uniform_real_distribution<double>
156 distribution(0, static_cast<double>(m_populationSize));
157
158 double minCost = m_minCostPerAgent[0];
159 std::size_t bestAgentIndex = 0;
160
161 for (std::size_t x = 0; x < m_populationSize; x++)
162 {
163 // For x in population select 3 random agents (a, b, c) different from x
164 std::size_t a = x;
165 std::size_t b = x;
166 std::size_t c = x;
167
168 // Agents must be different from each other and from x
169 while (a == x || b == x || c == x || a == b || a == c || b == c)
170 {
171 a = static_cast<std::size_t>(distribution(m_generator));
172 b = static_cast<std::size_t>(distribution(m_generator));
173 c = static_cast<std::size_t>(distribution(m_generator));
174 }
175
176 // Form intermediate solution z
177 std::vector<double> z(m_numberOfParameters);
178 for (std::size_t i = 0; i < m_numberOfParameters; i ++)
179 {
180 z[i] = m_population[a][i] + m_F * (m_population[b][i] - m_population[c][i]);
181 }
182
183 // Chose random R
184 std::uniform_real_distribution<double>
185 distributionParam(0, static_cast<double>(m_numberOfParameters));
186 std::size_t R = static_cast<std::size_t>(distributionParam(m_generator));
187
188 // Chose random r for each dimension
189 std::vector<double> r(m_numberOfParameters);
190 std::uniform_real_distribution<double> distributionPerX(0, 1);
191 for (auto& var : r)
192 {
193 var = distributionPerX(m_generator);
194 }
195
196 std::vector<double> newX(m_numberOfParameters);
197
198 // Execute crossing
199 for (std::size_t i = 0; i < m_numberOfParameters; i++)
200 {
201 if (r[i] < m_CR || i == R)
202 {
203 newX[i] = z[i];
204 }
205 else
206 {
207 newX[i] = m_population[x][i];
208 }
209 }
210
211 // Check if newX candidate satisfies constraints and skip it if not.
212 // If agent is skipped loop iteration x is decreased so that it is ensured
213 // that the population has constant size (equal to m_populationSize).
214 if (m_shouldCheckConstraints && !CheckConstraints(newX))
215 {
216 x--;
217 continue;
218 }
219
220 // Calculate new cost and decide should the newX be kept.
221 double newCost = m_cost.EvaluateCost(newX);
222 if (newCost < m_minCostPerAgent[x])
223 {
224 m_population[x] = newX;
225 m_minCostPerAgent[x] = newCost;
226 }
227
228 // Track the global best agent.
229 if (m_minCostPerAgent[x] < minCost)
230 {
231 minCost = m_minCostPerAgent[x];
232 bestAgentIndex = x;
233 }
234 }
235
236 m_minCost = minCost;
237 m_bestAgentIndex = bestAgentIndex;
238 }
