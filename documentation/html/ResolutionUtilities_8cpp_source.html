45 double CalculateResolution(const Tracks &Particles,
46 const RadiatorCell *radiatorCell,
47 const RadiatorArray &radiatorArray,
48 std::size_t Seed,
49 bool IncludeCentrePenalty) {
50 // Trace the photons of each track in parallel, one result per track
51 std::vector<ResolutionStruct> TrackResults(Particles.size());
52 const int NumberThreads = GetNumberThreads();
53 #pragma omp parallel for num_threads(NumberThreads)
54 for(std::size_t i = 0; i < Particles.size(); i++) {
55 // Each track gets its own photon sequence, independent of the thread schedule
56 Utilities::Random().SetSeed(Utilities::TrackSeed(Seed, i));
57 TrackResults[i] =
58 Utilities::TrackPhotons(Particles[i], *radiatorCell, radiatorArray);
59 }
60 // Put it all together, in track order so that the sums do not depend on the thread schedule
61 ResolutionStruct Total{};
62 std::size_t PhotonsHitWallTracks = 0;
63 for(const auto &resolutionStruct : TrackResults) {
64 if(!resolutionStruct.HitCorrectCell) {
65 continue;
66 }
67 if(resolutionStruct.HitTopWall) {
68 PhotonsHitWallTracks++;
69 } else if(resolutionStruct.N <= 1) {
70 PhotonsHitWallTracks++;
71 } else {
72 Total.x += resolutionStruct.x;
73 Total.N++;
74 Total.CentreHitDistance += resolutionStruct.CentreHitDistance;
75 }
76 }
77 if(Total.N == 0) {
78 return 1000.0;
79 }
80 // 0.0003 is the pixel size angular resolution
81 const double Resolution = Total.x/static_cast<double>(Total.N) + 0.0003*0.0003;
82 const auto NumberParticles = Particles.size();
83 auto GetFinalResolution = [&] (bool IncludeCentrePenalty) {
84 // Penalty when hitting the upper wall
85 const double WallPenalty = (10.0*static_cast<double>(PhotonsHitWallTracks))/
86 static_cast<double>(NumberParticles);
87 const double ResolutionWithPenalty = Resolution + WallPenalty;
88 if(IncludeCentrePenalty) {
89 // Penalty when far from the mirror centre
90 const Vector AverageRingPosition = Total.CentreHitDistance/Total.N;
91 const double CentrePenalty = 0.01*TMath::Sqrt(AverageRingPosition.Mag2());
92 return ResolutionWithPenalty + CentrePenalty;
93 } else {
94 return ResolutionWithPenalty;
95 }
96 };
97 const double FinalResolution = GetFinalResolution(IncludeCentrePenalty);
98 return FinalResolution;
99 }

239 de.Optimize(Iterations, true);
240 auto Result = de.GetBestAgent();
241 auto FixedParameters = resolutionOptimisable.GetFixedParameters();
242 std::size_t TotalParameters = Result.size() + FixedParameters.size();
243 std::vector<double> AllParameters;
244 constexpr std::array<std::string_view, 5> ParameterNames{
245 "Curvature",
246 "XPosition",
247 "ZPosition",
248 "DetPosition",
249 "DetTilt"};
250 std::size_t j = 0;
251 for(std::size_t i = 0; i < TotalParameters; i++) {
252 auto iter = FixedParameters.find(i);
253 if(iter != FixedParameters.end()) {
254 AllParameters.push_back(iter->second);
255 } else {
256 AllParameters.push_back(Result[j]);
257 j++;
258 }
259 }
260 std::string ResultFilename = Settings::GetString("Optimisation/Filename");
261 std::ofstream File(ResultFilename);
262 const std::string BarrelOrEndcap = Settings::GetString("General/BarrelOrEndcap");
263 const std::string Prefix = BarrelOrEndcap == "Barrel" ? "" : "EndCap";
264 for(std::size_t i = 0; i < TotalParameters; i++) {
265 File << Prefix;
266 File << "Radiator_c" << Column << "_r" << Row << "_" << ParameterNames[i] << " ";
267 File << AllParameters[i] << "\n";
268 }
269 //File << "\n" << "OptimalResolution: " << de.GetBestCost()*1000.0 << " mrad" << "\n";
270 File.close();
271 }
