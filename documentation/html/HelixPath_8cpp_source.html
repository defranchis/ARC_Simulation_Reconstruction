21Vector HelixPath::GetPosition(double s) const {
22 s -= m_Origin_s;
23 const double Sin = TMath::Sin(s*m_InvRadius);
24 const double Cos = TMath::Cos(s*m_InvRadius);
25 const double x = m_BFieldOn ?
26 m_Radius*(m_n.X()*Sin + m_n.Y()*(1 - Cos)) :
27 m_n.X()*s;
28 const double y = m_BFieldOn ?
29 m_Radius*(m_n.Y()*Sin - m_n.X()*(1 - Cos)) :
30 m_n.Y()*s;
31 const double z = m_n.Z()*s;
32 return m_Origin + Vector(x, y, z);
33}

35Vector HelixPath::GetDirection(double s) const {
36 s -= m_Origin_s;
37 const double Sin = TMath::Sin(s*m_InvRadius);
38 const double Cos = TMath::Cos(s*m_InvRadius);
39 const double x = m_BFieldOn ?
40 m_n.X()*Cos + m_n.Y()*Sin :
41 m_n.X();
42 const double y = m_BFieldOn ?
43 m_n.Y()*Cos - m_n.X()*Sin :
44 m_n.Y();
45 const double z = m_n.Z();
46 return Vector(x, y, z);
47}

49double HelixPath::SolvePathLength(const HelixFunctor &Functor,
50 double Min, double Max) const {
51 auto fcn = [&](double s) {
52 return Functor(GetPosition(s));
53 };
54 ROOT::Math::RootFinder Solver;
55 if(!Solver.Solve(fcn, Min, Max)) {
56 return -999.0;
57 }
58 return Solver.Root();
59}
