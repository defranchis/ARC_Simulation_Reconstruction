35PhotonHit SiPM::AddPhotonHit(Photon &photon) const {
36 const Vector DetectorCentre(m_DetectorPositionX,
37 0.0,
38 m_DetectorPositionZ);
39 const auto PhotonPosition = photon.GetPosition().LocalVector();
40 const Vector PhotonXZPosition(PhotonPosition.X(),
41 0.0,
42 PhotonPosition.Z());
43 PhotonHit photonHit{PhotonPosition,
44 &photon,
45 PhotonXZPosition - DetectorCentre};
46 if(photon.GetStatus() != Photon::Status::MirrorHit) {
47 return photonHit;
48 }
49 const double PhotonLambda = 1239.8/photon.GetEnergy();
50 if(PhotonLambda < m_Lambda[0] || PhotonLambda > m_Lambda[15]) {
51 std::cout << "Warning! Generated photon outside energy range of SiPM!\n";
52 }
53 if(!IsDetectorHit(photon)) {
54 photon.UpdatePhotonStatus(Photon::Status::DetectorMiss);
55 return photonHit;
56 }
57 const double RandomNumber = Utilities::Random().Uniform(0.0, m_MaxPDE);
58 const double Efficiency = m_Interpolator.Eval(PhotonLambda)*0.90*0.80;
59 if(RandomNumber <= Efficiency) {
60 photon.UpdatePhotonStatus(Photon::Status::DetectorHit);
61 } else {
62 photon.UpdatePhotonStatus(Photon::Status::EfficiencyMiss);
63 }
64 return photonHit;
65}

67void SiPM::PlotHits(const std::string &Filename,
68 const std::vector<PhotonHit> &photonHits) const {
69 std::vector<double> xHits_Aerogel, yHits_Aerogel, xHits_Gas, yHits_Gas;
70 for(const auto &PhotonHit : photonHits) {
71 if(PhotonHit.m_Photon->GetRadiator() == Photon::Radiator::Aerogel) {
72 xHits_Aerogel.push_back(PhotonHit.m_HitPosition.X()*100.0);
73 yHits_Aerogel.push_back(PhotonHit.m_HitPosition.Y()*100.0);
74 } else if(PhotonHit.m_Photon->GetRadiator() == Photon::Radiator::Gas) {
75 xHits_Gas.push_back(PhotonHit.m_HitPosition.X()*100.0);
76 yHits_Gas.push_back(PhotonHit.m_HitPosition.Y()*100.0);
77 } else {
78 std::cout << "Warning! Detector hit with unknown origin\n";
79 }
80 }
81 TGraph Graph_Aerogel(static_cast<int>(xHits_Aerogel.size()),
82 xHits_Aerogel.data(),
83 yHits_Aerogel.data());
84 TGraph Graph_Gas(static_cast<int>(xHits_Gas.size()),
85 xHits_Gas.data(),
86 yHits_Gas.data());
87 TCanvas c("c", "", 1200, 1200);
88 Graph_Aerogel.SetMarkerStyle(kFullDotLarge);
89 Graph_Gas.SetMarkerStyle(kFullDotLarge);
90 Graph_Aerogel.SetMarkerColor(kBlue);
91 Graph_Gas.SetMarkerColor(kRed);
92 Graph_Aerogel.SetTitle("Photon hits;x (cm); y (cm)");
93 Graph_Aerogel.Draw("AP");
94 gPad->SetLeftMargin(0.15f);
95 gPad->SetBottomMargin(0.15f);
96 Graph_Aerogel.GetXaxis()->SetLimits((m_DetectorPositionX
97 - m_DetectorSizeX/2.0)*100.0,
98 (m_DetectorPositionX
99 + m_DetectorSizeX/2.0)*100.0);
100 Graph_Aerogel.SetMinimum((m_DetectorPositionY - m_DetectorSizeY/2.0)*100.0);
101 Graph_Aerogel.SetMaximum((m_DetectorPositionY + m_DetectorSizeY/2.0)*100.0);
102 Graph_Aerogel.Draw("AP");
103 Graph_Gas.Draw("P SAME");
104 TLegend Legend(0.7, 0.8, 0.9, 0.9);
105 // The momentum is a double setting, the legend rounds it to whole GeV
106 const int Momentum = static_cast<int>(Settings::GetDouble("Particle/Momentum"));
107 const std::string AerogelLabel = "Aerogel " + std::to_string(Momentum) + " GeV";
108 Legend.AddEntry(&Graph_Aerogel, AerogelLabel.c_str(), "P");
109 const std::string GasLabel = "Gas " + std::to_string(Momentum) + " GeV";
110 Legend.AddEntry(&Graph_Gas, GasLabel.c_str(), "P");
111 Legend.Draw();
112 c.Draw();
113 c.SaveAs(Filename.c_str());
114}

116std::unique_ptr<TObject> SiPM::DrawSiPM(const Vector &RadiatorPosition) const {
117 const double PositionZ = RadiatorPosition.X()
118 + (m_EndCap ? m_DetectorPositionX : m_DetectorPositionZ);
119 const double PositionX = RadiatorPosition.Z()
120 + (m_EndCap ? m_DetectorPositionZ : m_DetectorPositionX);
121 std::array<double, 5> xPoints, zPoints;
122 const double DetectorTilt = m_EndCap ?
123 -m_DetectorTilt + TMath::Pi()/2.0 :
124 m_DetectorTilt;
125 const double CosTheta = TMath::Cos(DetectorTilt);
126 const double SinTheta = TMath::Sin(DetectorTilt);
127 // Lower left corner
128 xPoints[0] = PositionX - CosTheta*m_DetectorSizeX/2.0;
129 zPoints[0] = PositionZ - 0.002 - SinTheta*m_DetectorSizeX/2.0;
130 // Lower right corner
131 xPoints[1] = PositionX + CosTheta*m_DetectorSizeX/2.0;
132 zPoints[1] = PositionZ -0.002 + SinTheta*m_DetectorSizeX/2.0;
133 // Upper right corner
134 xPoints[2] = PositionX + CosTheta*m_DetectorSizeX/2.0;
135 zPoints[2] = PositionZ - 0.0 + SinTheta*m_DetectorSizeX/2.0;
136 // Upper left corner
137 xPoints[3] = PositionX - CosTheta*m_DetectorSizeX/2.0;
138 zPoints[3] = PositionZ - 0.0 - SinTheta*m_DetectorSizeX/2.0;
139 // Back to left corner
140 xPoints[4] = PositionX - CosTheta*m_DetectorSizeX/2.0;
141 zPoints[4] = PositionZ - 0.002 - SinTheta*m_DetectorSizeX/2.0;
142 TPolyLine DetectorLine(5, xPoints.data(), zPoints.data());
143 DetectorLine.SetFillColor(kBlack);
144 return std::make_unique<TPolyLine>(DetectorLine);
145}

147bool SiPM::IsDetectorHit(const Photon &photon) const {
148 if(!photon.GetRadiatorCell()->IsInsideCell(photon)) {
149 return false;
150 }
151 const auto PhotonPosition = photon.GetPosition().LocalVector();
152 const double photon_x = PhotonPosition.X();
153 const double photon_z = PhotonPosition.Z();
154 const double XDist = TMath::Sqrt((photon_x - m_DetectorPositionX)*
155 (photon_x - m_DetectorPositionX)
156 + (photon_z - m_DetectorPositionZ)*
157 (photon_z - m_DetectorPositionZ));
158 if(XDist > m_DetectorSizeX/2.0) {
159 return false;
160 } else if(TMath::Abs(PhotonPosition.Y() - m_DetectorPositionY)
161 > m_DetectorSizeY/2.0) {
162 return false;
163 } else {
164 return true;
165 }
166}

172void SiPM::SetDetectorTilt(double Angle) {
173 m_DetectorTilt = Angle;
174 const double AbsAngle = TMath::Abs(Angle);
175 const double DetectorSizeXOver2 = m_DetectorSizeX*0.5;
176 if(AbsAngle > TMath::ASin(m_CoolingThickness/DetectorSizeXOver2)) {
177 m_DetectorPositionZ = TMath::Sin(AbsAngle)*DetectorSizeXOver2;
178 m_DetectorPositionZ -= m_CoolingThickness;
179 } else {
180 m_DetectorPositionZ = 0.0;
181 }
182}
